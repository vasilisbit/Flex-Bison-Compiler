public class SwitchCaseExample {
    public void main() {
        int day = 3;
        switch (day) {
            case 1:
                out.print("Monday");
                break;
            case 2:
                out.print("Tuesday");
                break;
            case 3:
                out.print("Wednesday");
                break;
            case 4:
                out.print("Thursday");
                break;
            case 5:
                out.print("Friday");
                break;
            default:
                out.print("Weekend");
        }
    }
}
